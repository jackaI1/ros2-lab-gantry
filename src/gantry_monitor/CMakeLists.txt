cmake_minimum_required(VERSION 3.16)
project(gantry_monitor)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(trajectory_safety_monitor src/monitor.cpp)
target_include_directories(trajectory_safety_monitor PRIVATE include)
target_compile_features(trajectory_safety_monitor PRIVATE cxx_std_17)
target_compile_options(trajectory_safety_monitor PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(trajectory_safety_monitor rclcpp sensor_msgs std_msgs)
install(TARGETS trajectory_safety_monitor DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_limits test/test_limits.cpp)
  target_include_directories(test_limits PRIVATE include)
  target_compile_features(test_limits PRIVATE cxx_std_17)
endif()
ament_package()
