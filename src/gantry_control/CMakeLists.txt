cmake_minimum_required(VERSION 3.16)
project(gantry_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/protocol_server scripts/run_plate scripts/pendant DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_pytest_test(protocol_unit test/test_protocol.py)
  ament_add_pytest_test(pendant_unit test/test_pendant.py)
  ament_add_pytest_test(motion_unit test/test_motion.py)
  ament_add_pytest_test(triage_unit test/test_triage.py)
  ament_add_pytest_test(journal_safety_unit test/test_journal_safety.py)
  add_launch_test(test/test_simulation.py TIMEOUT 420)
endif()
ament_package()
