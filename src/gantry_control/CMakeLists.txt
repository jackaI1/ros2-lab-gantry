cmake_minimum_required(VERSION 3.16)
project(gantry_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/protocol_server scripts/run_plate DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_pytest_test(protocol_unit test/test_protocol.py)
  add_launch_test(test/test_simulation.py TIMEOUT 240)
endif()
ament_package()
